221 template <typename T> class SubscriptionHandler
222 : public ISubscriptionHandler
223 {
224 // Documentation inherited.
225 public: explicit SubscriptionHandler(const std::string &_pUuid,
226 const std::string &_nUuid,
227 const SubscribeOptions &_opts = SubscribeOptions())
228 : ISubscriptionHandler(_pUuid, _nUuid, _opts)
229 {
230 // Parse straight from the buffer (no intermediate string copy).
231 this->SetMsgFactory(
232 [](const char *_data, std::size_t _size,
233 const std::string &/*_type*/) -> std::shared_ptr<ProtoMsg>
234 {
235 auto msgPtr = std::make_shared<T>();
236 if (!msgPtr->ParseFromArray(_data, static_cast<int>(_size)))
237 {
238 std::cerr << "SubscriptionHandler::CreateMsgFromBuffer() error: "
239 << "ParseFromArray failed" << std::endl;
240 }
241 return msgPtr;
242 });
243 }
244
245 // Documentation inherited.
246 public: const std::shared_ptr<ProtoMsg> CreateMsg(
247 const std::string &_data,
248 const std::string &_type) const
249 {
250 // Safe to delegate: the constructor registered a factory.
251 return this->CreateMsgFromBuffer(_data.data(), _data.size(), _type);
252 }
253
254 // Documentation inherited.
255 public: std::string TypeName()
256 {
257 return std::string(T().GetTypeName());
258 }
259
262 public: void SetCallback(const MsgCallback<T> &_cb)
263 {
264 this->cb = _cb;
265 }
266
267#ifdef HAVE_ZENOH
272 public: void SetCallback(const MsgCallback<T> &_cb,
273 std::shared_ptr<zenoh::Session> _session,
274 const FullyQualifiedTopic &_fullyQualifiedTopic)
275 {
276 this->SetCallback(std::move(_cb));
277 this->CreateLivelinessToken(_session, _fullyQualifiedTopic);
278 }
279#endif
280
281 // Documentation inherited.
282 public: bool RunLocalCallback(const ProtoMsg &_msg,
283 const MessageInfo &_info)
284 {
285 // No callback stored.
286 if (!this->cb)
287 {
288 std::cerr << "SubscriptionHandler::RunLocalCallback() error: "
289 << "Callback is NULL" << std::endl;
290 return false;
291 }
292
293 // Check the subscription throttling option.
294 if (!this->UpdateThrottling())
295 return true;
296
297#if GOOGLE_PROTOBUF_VERSION >= 5028000
298 auto msgPtr = google::protobuf::DynamicCastMessage<T>(&_msg);
299#elif GOOGLE_PROTOBUF_VERSION >= 4022000
300 auto msgPtr = google::protobuf::internal::DownCast<const T*>(&_msg);
301#elif GOOGLE_PROTOBUF_VERSION >= 3000000
302 auto msgPtr = google::protobuf::down_cast<const T*>(&_msg);
303#else
304 auto msgPtr = google::protobuf::internal::down_cast<const T*>(&_msg);
305#endif
306
307 // Verify the dynamically casted message is valid
308 if (msgPtr == nullptr)
309 {
310 if (_msg.GetDescriptor() != nullptr)
311 {
312 std::cerr << "SubscriptionHandler::RunLocalCallback() error: "
313 << "Failed to cast the message of the type "
314 << _msg.GetDescriptor()->full_name()
315 << " to the specified type" << '\n';
316 }
317 else
318 {
319 std::cerr << "SubscriptionHandler::RunLocalCallback() error: "
320 << "Failed to cast the message of an unknown type"
321 << " to the specified type" << '\n';
322 }
323 std::cerr.flush();
324 return false;
325 }
326
327 this->cb(*msgPtr, _info);
328 return true;
329 }
330
332 private: MsgCallback<T> cb;
333 };

337 template <> class SubscriptionHandler<ProtoMsg>
338 : public ISubscriptionHandler
339 {
340 // Documentation inherited.
341 public: explicit SubscriptionHandler(const std::string &_pUuid,
342 const std::string &_nUuid,
343 const SubscribeOptions &_opts = SubscribeOptions())
344 : ISubscriptionHandler(_pUuid, _nUuid, _opts)
345 {
346 // Look the type up at runtime and parse straight from the buffer
347 // (no intermediate string copy).
348 this->SetMsgFactory(
349 [](const char *_data, std::size_t _size,
350 const std::string &_type) -> std::shared_ptr<ProtoMsg>
351 {
352 std::shared_ptr<google::protobuf::Message> msgPtr;
353
354 const google::protobuf::Descriptor *desc =
355 google::protobuf::DescriptorPool::generated_pool()
356 ->FindMessageTypeByName(_type);
357
358 // First, check if we have the descriptor from the generated
359 // proto classes.
360 if (desc)
361 {
362 msgPtr.reset(
363 google::protobuf::MessageFactory::generated_factory()
364 ->GetPrototype(desc)->New());
365 }
366 else
367 {
368 // Fallback on Gazebo Msgs if the message type is not found.
369 msgPtr = gz::msgs::Factory::New(_type);
370 }
371
372 if (!msgPtr)
373 return nullptr;
374
375 if (!msgPtr->ParseFromArray(_data, static_cast<int>(_size)))
376 {
377 std::cerr << "CreateMsgFromBuffer() error: ParseFromArray failed"
378 << std::endl;
379 return nullptr;
380 }
381
382 return msgPtr;
383 });
384 }
385
386 // Documentation inherited.
387 public: const std::shared_ptr<ProtoMsg> CreateMsg(
388 const std::string &_data,
389 const std::string &_type) const
390 {
391 // Safe to delegate: the constructor registered a factory.
392 return this->CreateMsgFromBuffer(_data.data(), _data.size(), _type);
393 }
394
395 // Documentation inherited.
396 public: std::string TypeName()
397 {
398 return kGenericMessageType;
399 }
400
403 public: void SetCallback(const MsgCallback<ProtoMsg> &_cb)
404 {
405 this->cb = _cb;
406 }
407
408#ifdef HAVE_ZENOH
413 public: void SetCallback(const MsgCallback<ProtoMsg> &_cb,
414 std::shared_ptr<zenoh::Session> _session,
415 const FullyQualifiedTopic &_fullyQualifiedTopic)
416 {
417 this->SetCallback(std::move(_cb));
418 this->CreateLivelinessToken(_session, _fullyQualifiedTopic);
419 }
420#endif
421
422 // Documentation inherited.
423 public: bool RunLocalCallback(const ProtoMsg &_msg,
424 const MessageInfo &_info)
425 {
426 // No callback stored.
427 if (!this->cb)
428 {
429 std::cerr << "SubscriptionHandler::RunLocalCallback() "
430 << "error: Callback is NULL" << std::endl;
431 return false;
432 }
433
434 // Check the subscription throttling option.
435 if (!this->UpdateThrottling())
436 return true;
437
438 this->cb(_msg, _info);
439 return true;
440 }
