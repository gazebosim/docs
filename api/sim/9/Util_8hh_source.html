295 template <class ComponentType>
296 bool enableComponent(EntityComponentManager &_ecm,
297 Entity _entity,
298 bool _enable = true,
299 const ComponentType &_comp = ComponentType())
300 {
301 bool changed{false};
302
303 auto exists = _ecm.Component<ComponentType>(_entity);
304 if (_enable && !exists)
305 {
306 _ecm.CreateComponent(_entity, _comp);
307 changed = true;
308 }
309 else if (!_enable && exists)
310 {
311 _ecm.RemoveComponent<ComponentType>(_entity);
312 changed = true;
313 }
314 return changed;
315 }
